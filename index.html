Siap Mengumpulkan LKPD?

Jawaban akan dikirim ke Google Sheet Guru untuk dinilai sesuai Rubrik Bagian D.

Panel Guru: Integrasi Google Sheets & Rubrik Penilaian (Bagian D)

Hubungkan URL Google Apps Script dan nilai tugas siswa langsung.

Tempelkan URL Web App Google Apps Script Anda di bawah ini:

Kode `Code.gs` (Otomatis Membuat Header + Kolom Rubrik D + Rumus Total): 
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var doc = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = doc.getSheetByName('Data_LKPD_BPUPKI') || doc.insertSheet('Data_LKPD_BPUPKI');
    if (sheet.getLastRow() === 0) {
      var headers = [
        'Waktu', 'Nama Siswa', 'Kelas', 'Kelompok', 'Tanggal',
        'A1 Soepomo', 'A1 Soekarno', 'A1 Yamin', 'A1 Soepomo Rumusan', 'A1 Soekarno Rumusan',
        'A1.1 Analisis', 'A1.2 Analisis', 'A2 1', 'A2 2', 'A2 3', 'A2 4', 'A2 5', 'A2 6',
        'A3 Panitia 9', 'A3 Sila 1', 'A4 Refleksi',
        '[RUBRIK D1] Komparasi (30)', '[RUBRIK D2] Literasi (25)', '[RUBRIK D3] Kompromi (25)', '[RUBRIK D4] Refleksi (20)',
        '[TOTAL SKOR AKHIR] (100)', 'Catatan Guru'
      ];
      sheet.appendRow(headers);
      sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#1c1917').setFontColor('#fef3c7');
      sheet.setFrozenRows(1);
    }
    var data = JSON.parse(e.postData.contents);
    var nextRow = sheet.getLastRow() + 1;
    var totalFormula = '=SUM(V' + nextRow + ':Y' + nextRow + ')';
    sheet.appendRow([
      new Date(), data.studentName, data.studentClass, data.studentGroup, data.activityDate,
      data.soepomoMeta, data.soekarnoMeta, data.yaminRumusan, data.soepomoRumusan, data.soekarnoRumusan,
      data.a1_analisis_1, data.a1_analisis_2, data.a2_1, data.a2_2, data.a2_3, data.a2_4, data.a2_5, data.a2_6,
      data.a3_panitia9, data.a3_perubahan_sila1, data.a4_refleksi,
      '', '', '', '', totalFormula, ''
    ]);
    return ContentService.createTextOutput(JSON.stringify({ status: 'success' })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  } finally { lock.releaseLock(); }
}
